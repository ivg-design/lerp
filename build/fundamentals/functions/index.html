Lesson 1.3: Functions & Closures

Learning Objectives​

  • Define and call functions with parameters
  • Use return values (including multiple returns)
  • Understand scope and upvalues
  • Create closures for stateful behavior
  • Use variadic functions

Basic Functions​

-- Function with type annotations
local function add(a: number, b: number): number
    return a + b
end

-- Calling the function
local result = add(5, 3)  -- 8

Multiple Return Values​

Luau functions can return multiple values:

local function minMax(a: number, b: number): (number, number)
    return math.min(a, b), math.max(a, b)
end

local min, max = minMax(5, 10)
print(min, max)  -- 5, 10

Anonymous Functions​

local greet = function(name: string): string
    return `Hello, {name}!`
end

-- Often used for callbacks
table.sort(items, function(a, b)
    return a.score > b.score
end)

Scope and Upvalues​

Module-Level Variables​

local globalCounter = 0  -- Module scope

local function incrementGlobal(): number
    globalCounter += 1
    return globalCounter
end

print(incrementGlobal())  -- 1
print(incrementGlobal())  -- 2
print(incrementGlobal())  -- 3

Upvalues​

Inner functions can access variables from outer scopes:

local function outer()
    local outerValue = "outer"

    local function inner()
        -- 'outerValue' is an upvalue here
        return outerValue .. " + inner"
    end

    return inner()
end

print(outer())  -- "outer + inner"

Closures​

A closure is a function that captures variables from its surrounding scope:

local function makeCounter()
    local count = 0  -- Captured by the returned function

    return function()
        count += 1
        return count
    end
end

local counter = makeCounter()
print(counter())  -- 1
print(counter())  -- 2
print(counter())  -- 3

-- Each call to makeCounter creates independent state
local counter2 = makeCounter()
print(counter2())  -- 1 (independent!)

Factory Functions​

Closures enable the factory pattern:

local function createMultiplier(factor: number)
    return function(x: number): number
        return x * factor
    end
end

local double = createMultiplier(2)
local triple = createMultiplier(3)

print(double(5))   -- 10
print(triple(5))   -- 15
print(double(10))  -- 20

Practical Patterns​

Timer Factory:

local function createTimer(duration: number)
    local elapsed = 0
    return function(dt: number): boolean
        elapsed += dt
        return elapsed >= duration
    end
end

local timer = createTimer(2.0)  -- 2 second timer
-- In advance():
if timer(seconds) then
    print("Timer finished!")
end

Memoization:

local function memoize(fn)
    local cache = {}
    return function(x)
        if cache[x] == nil then
            cache[x] = fn(x)
        end
        return cache[x]
    end
end

local expensiveCalc = memoize(function(n)
    -- Only calculated once per unique n
    return n * n * n
end)

Variadic Functions​

Functions that accept variable numbers of arguments:

local function sum(...): number
    local args = {...}
    local total = 0
    for _, v in args do
        total += v
    end
    return total
end

print(sum(1, 2, 3))        -- 6
print(sum(10, 20, 30, 40)) -- 100

Using select()​

local function average(...): number
    local count = select("#", ...)  -- Get argument count
    local total = sum(...)
    return total / count
end

print(average(10, 20, 30))  -- 20

Practice Exercises​

Exercise 1: Basic Functions​

-- Create these utility functions:

-- 1. calculateDamage(base, multiplier, reduction)
-- Returns: (base * multiplier) - reduction
-- If result < 0, return 0
local function calculateDamage(base, multiplier, reduction)
    local result = (base * multiplier) - reduction
    return if result < 0 then 0 else result
end

-- 2. isInRange(value, min, max)
-- Returns: true if value is between min and max (inclusive)
local function isInRange(value, min, max)
    return value >= min and value <= max
end

-- Test:
print(calculateDamage(50, 2, 30))  -- 70
print(calculateDamage(10, 1, 50))  -- 0
print(isInRange(5, 1, 10))         -- true
print(isInRange(15, 1, 10))        -- false

Exercise 2: Multiple Returns​

-- divmod(dividend, divisor) -> quotient, remainder
local function divmod(dividend, divisor)
    return dividend // divisor, dividend % divisor
end

local q, r = divmod(17, 5)
print(`17 / 5 = {q} remainder {r}`)  -- 3 remainder 2

-- getStats(values) -> min, max, sum, average
local function getStats(values)
    local min, max = values[1], values[1]
    local sum = 0
    for _, v in values do
        if v < min then min = v end
        if v > max then max = v end
        sum += v
    end
    return min, max, sum, sum / #values
end

local min, max, sum, avg = getStats({10, 20, 30, 40, 50})
print(`Min: {min}, Max: {max}, Sum: {sum}, Avg: {avg}`)

Exercise 3: Closures​

-- makeAccumulator: adds to running total
local function makeAccumulator()
    local total = 0
    return function(n)
        total += n
        return total
    end
end

local acc = makeAccumulator()
print(acc(10))  -- 10
print(acc(5))   -- 15
print(acc(25))  -- 40

-- makeToggle: toggles boolean state
local function makeToggle(initial)
    local state = initial
    return function()
        state = not state
        return state
    end
end

local toggle = makeToggle(false)
print(toggle())  -- true
print(toggle())  -- false
print(toggle())  -- true

Challenge: Function Composition​

-- compose(f, g) returns a function where compose(f,g)(x) = f(g(x))
local function compose(f, g)
    return function(x)
        return f(g(x))
    end
end

local function addOne(x) return x + 1 end
local function double(x) return x * 2 end

local addThenDouble = compose(double, addOne)
print(addThenDouble(5))  -- double(addOne(5)) = double(6) = 12

local doubleThenAdd = compose(addOne, double)
print(doubleThenAdd(5))  -- addOne(double(5)) = addOne(10) = 11

Key Patterns to Remember​

  1. Factory Pattern: Functions that return configured functions
  2. Closure State: Captured variables persist across calls
  3. Multiple Returns: Luau's elegant way to return related values
  4. Variadic Functions: Use ... for flexible argument counts

Self-Assessment Checklist​

  • I can define functions with typed parameters and return types
  • I can return multiple values from a function
  • I understand how closures capture variables from outer scope
  • I can create factory functions that return other functions
  • I can use ... for variadic functions
